float atr      = ta.atr(atrLen)
float ema      = ta.ema(close, emaLen)
float vol      = nz(volume)
bool  hasVol   = ta.cum(vol) > 0
float vMa      = ta.sma(vol, 50)
float vSd      = ta.stdev(vol, 50)
float vS10     = ta.sma(vol, 10)
float volZ     = hasVol and vSd > 0 ? (vol - vMa) / vSd : 0.0
float volRel   = hasVol and vMa > 0 ? vol / vMa : 1.0
float rng      = high - low
float body     = math.abs(close - open)
float lowestL  = ta.lowest(low, climaxLook)
float highestH = ta.highest(high, climaxLook)
float rngMa    = ta.sma(rng, 20)
bool  bigBar   = rng >= climaxAtr * atr
bool  climaxV  = hasVol ? volZ >= climaxZ : rng >= climaxAtr * 1.6 * atr
bool  isSC     = not na(atr) and close < ema and low <= lowestL and bigBar and climaxV
bool  isBC     = not na(atr) and close > ema and high >= highestH and bigBar and climaxV

f_htfBias() =>
    float e = ta.ema(close, 50)
    close > e and e > e[5] ? 1 : close < e and e < e[5] ? -1 : 0

f_x(int x) =>
    math.max(x, bar_index - 4900)

f_cleanup(int cut) =>
    while qL.size() > 0 and qLi.first() <= cut
        label.delete(qL.shift())
        qLi.shift()
    while qN.size() > 0 and qNi.first() <= cut
        line.delete(qN.shift())
        qNi.shift()
    while qB.size() > 0 and qBi.first() <= cut
        box.delete(qB.shift())
        qBi.shift()

f_ev(int x, float y, string s, string f, bool below, color c, string tip) =>
    if showLabels
        label l = label.new(f_x(x), y, lblStyle == "Short" ? s : f, xloc = xloc.bar_index, style = below ? label.style_label_up : label.style_label_down, color = c, textcolor = color.white, size = sz, tooltip = tip)
        qL.push(l)
        qLi.push(schId)

f_phase(int x, string t, float top, float bot) =>
    if showPhases
        line ln = line.new(f_x(x), top, f_x(x), bot, color = color.new(chart.fg_color, 55), style = line.style_dotted, width = 1)
        qN.push(ln)
        qNi.push(schId)
        label lb = label.new(f_x(x), top, t, xloc = xloc.bar_index, style = label.style_label_down, color = color.new(chart.bg_color, 100), textcolor = color.new(chart.fg_color, 20), size = size.small)
        qL.push(lb)
        qLi.push(schId)

f_tip(string name, float lvl, float rel) =>
    name + "\nLevel: " + str.tostring(lvl, format.mintick) + (hasVol ? "\nVolume: " + str.tostring(rel, "0.0") + "x average" : "")

f_score(float cz, bool sp, int q, bool ts, float sz_, int life, bool lowEff, int hb, int dd) =>
    int s = 0
    s += cz >= climaxZ + 1.0 ? 15 : 8
    s += sp ? int(math.round(20.0 * q / 100.0)) : 0
    s += ts ? 15 : 0
    s += sz_ >= 1.0 ? 15 : sz_ >= sosZ ? 8 : 0
    s += life >= causeRef ? 15 : life >= causeRef / 2 ? 8 : 0
    s += lowEff ? 10 : 0
    s += hb == dd ? 10 : hb == 0 ? 5 : 0
    s

f_grade(int s) =>
    s >= 75 ? "A" : s >= 55 ? "B" : "C"

f_gIdx(string g) =>
    g == "A" ? 0 : g == "B" ? 1 : 2

f_phaseName(int s) =>
    s == 1 or s == 2 ? "A" : s == 3 ? "B" : s == 4 ? "C" : s == 5 ? "D" : s == 6 ? "E" : "—"

f_cause(int life) =>
    math.min(math.max(float(life) / causeRef, 1.0), causeMax)

f_pct(int a, int b) =>
    b > 0 ? str.tostring(100.0 * a / b, "0") + "%" : "—"

f_pos(string p) =>
    p == "Top Left" ? position.top_left : p == "Top Center" ? position.top_center : p == "Top Right" ? position.top_right : p == "Middle Left" ? position.middle_left : p == "Middle Center" ? position.middle_center : p == "Middle Right" ? position.middle_right : p == "Bottom Left" ? position.bottom_left : p == "Bottom Center" ? position.bottom_center : position.bottom_right

" · " + side : "") + " · " + str.tostring(px, format.mintick) + extra

// ═══════════════════════════════════════════════════════════════════════
// ENGINE (bar close only — no repaint)
// ═══════════════════════════════════════════════════════════════════════
if barstate.isconfirmed and not na(atr)
    bool   endNow   = false
    bool   endFail  = false
    bool   doBreak  = false
    string endTxt   = ""
    bool   goSOS    = false
    bool   doSpring = false
    bool   doLPS    = false
    bool   doTest   = false
    color  col      = d == 1 ? accCol : distCol
    float  xC       = d == 1 ? low : high
    float  xB       = d == 1 ? high : low
    float  H        = math.abs(bLvl - cLvl)
    string nmS      = d == 1 ? "Spring" : "UTAD"

    // ── trade management ───────────────────────────────────────────
    if inTrade and bar_index > tBar
        bool  hSL = tDir == 1 ? low <= tCurSL : high >= tCurSL
        bool  done = false
        string rTx = ""
        array<float> tps = array.from(tTP1, tTP2, tTP3)
        if hSL
            tReal   += tRemain * tDir * (tCurSL - tE) / tRisk
            tRemain := 0.0
            done    := true
            rTx     := tCurSL == tSL ? "SL" : tHits == 1 ? "BE" : "TP1 stop"
            evSL    := tCurSL == tSL
        else
            if scaleOut
                while tHits < 3 and (tDir == 1 ? high >= tps.get(tHits) : low <= tps.get(tHits))
                    float rk = math.abs(tps.get(tHits) - tE) / tRisk
                    tReal   += rk / 3.0
                    tRemain -= 1.0 / 3.0
                    tHits   += 1
                    if tHits == 1
                        tCurSL := tE
                        evTP1  := true
                    else if tHits == 2
                        tCurSL := tTP1
                        evTP2  := true
                    else
                        evTP3  := true
                if tHits >= 3
                    tRemain := 0.0
                    done    := true
                    rTx     := "TP3"
            else
                if tDir == 1 ? high >= tTP1 : low <= tTP1
                    if tHits == 0
                        evTP1 := true
                    tHits := math.max(tHits, 1)
                if tDir == 1 ? high >= tTP2 : low <= tTP2
                    tReal   += math.abs(tTP2 - tE) / tRisk
                    tRemain := 0.0
                    tHits   := 2
                    evTP2   := true
                    done    := true
                    rTx     := "TP2"
        if not done and bar_index - tBar >= tradeMax
            tReal   += tRemain * tDir * (close - tE) / tRisk
            tRemain := 0.0
            done    := true
            rTx     := "Time exit"
        if done
            int gi = f_gIdx(tGrade)
            gN.set(gi, gN.get(gi) + 1)
            gR.set(gi, gR.get(gi) + tReal)
            trT.push(time)
            trR.push(tReal)
            trH.push(tHits)
            inTrade := false
            if showTrade
                box.set_right(tRiskB, bar_index)
                box.set_right(tRewB, bar_index)
                line.set_x2(tTp1L, bar_index)
                line.set_x2(tTp2L, bar_index)
                string rs = rTx + " " + (tReal >= 0 ? "+" : "") + str.tostring(tReal, "0.0") + "R"
                f_ev(bar_index, tDir == 1 ? high : low, rs, rs, tDir == -1, tReal > 0 ? accCol : tReal < 0 ? distCol : color.gray, "Trade closed: " + rTx + "\nResult: " + str.tostring(tReal, "0.00") + "R\nTargets hit: " + str.tostring(tHits))
            if aExit
                msgs.push(f_msg("Trade closed " + rTx, tDir == 1 ? "long" : "short", close, aJson ? ",\"r\":" + str.tostring(tReal, "0.00") : ""))
            if st == 6 and tSch == schId
                endNow := true
                endTxt := "Trade closed (" + rTx + ")"
        else
            if showTrade
                box.set_right(tRiskB, bar_index)
                box.set_right(tRewB, bar_index)
                line.set_x2(tTp1L, bar_index)
                line.set_x2(tTp2L, bar_index)
            if aExit and (evTP1 or evTP2)
                msgs.push(f_msg(evTP2 ? "TP2 hit" : "TP1 hit", tDir == 1 ? "long" : "short", close, ""))

    // ── state machine ──────────────────────────────────────────────
    if st == 0
        if isSC or isBC
            d         := isSC ? 1 : -1
            st        := 1
            cLvl      := isSC ? low : high
            cBar      := bar_index
            cVol      := vol
            cZv       := volZ
            cRel      := volRel
            ext       := close
            extBar    := bar_index
            hadSpring := false
            hadTest   := false
            eShown    := false
            spQ       := 0
            reCnt     := 0
            pend      := 0
            lastEv    := isSC ? "SC" : "BC"
    else if st == 1
        float pen = d * (cLvl - xC)
        if pen > 0
            if pen <= atr
                cLvl   := xC
                cBar   := bar_index
                cVol   := math.max(cVol, vol)
                cZv    := math.max(cZv, volZ)
                cRel   := math.max(cRel, volRel)
                ext    := close
                extBar := bar_index
            else
                st := 0
        else
            if d * (xB - ext) > 0
                ext    := xB
                extBar := bar_index
            float h = math.abs(ext - cLvl)
            if extBar > cBar and d * (ext - close) >= arPull * atr and h >= minHAtr * atr
                if h <= maxHAtr * atr
                    bLvl  := ext
                    arBar := extBar
                    st    := 2
                    schId += 1
                    if d == 1
                        nAcc += 1
                    else
                        nDist += 1
                    f_cleanup(schId - keepN)
                    float top = math.max(cLvl, bLvl)
                    float bot = math.min(cLvl, bLvl)
                    float mid = (top + bot) / 2
                    rBox := na
                    midL := na
                    if showBox
                        rBox := box.new(f_x(cBar), top, bar_index, bot, border_color = color.new(col, 35), bgcolor = color.new(col, 91), text = (d == 1 ? "Accumulation" : "Distribution") + " · Phase A", text_color = color.new(col, 10), text_size = size.small, text_halign = text.align_left, text_valign = text.align_top)
                        qB.push(rBox)
                        qBi.push(schId)
                        midL := line.new(f_x(cBar), mid, bar_index, mid, color = color.new(col, 50), style = line.style_dashed)
                        qN.push(midL)
                        qNi.push(schId)
                    f_ev(cBar, cLvl, d == 1 ? "SC" : "BC", d == 1 ? "Selling Climax" : "Buying Climax", d == 1, col, f_tip(d == 1 ? "Selling Climax" : "Buying Climax", cLvl, cRel))
                    f_ev(arBar, bLvl, "AR", d == 1 ? "Automatic Rally" : "Automatic Reaction", d == -1, col, f_tip(d == 1 ? "Automatic Rally" : "Automatic Reaction", bLvl, volRel[bar_index - arBar]))
                    f_phase(cBar, "A", top, bot)
                    lastEv  := "AR"
                    evRange := true
                    if aRange
                        msgs.push(f_msg("New " + (d == 1 ? "accumulation" : "distribution") + " range", d == 1 ? "long bias" : "short bias", close, aJson ? ",\"rangeLow\":" + str.tostring(bot) + ",\"rangeHigh\":" + str.tostring(top) : ""))
                else
                    st := 0
            else if bar_index - cBar > arMax
                st := 0
    else if st == 2
        if d * (cLvl - close) > springPen * atr
            doBreak := true
        else if d * (close - bLvl) > H
            endNow := true
            endTxt := "Left range"
        else if bar_index - arBar >= 2 and d * (xC - cLvl) <= stZone * H and d * (close - cLvl) > 0 and (not hasVol or vol <= stEffort * cVol)
            st     := 3
            bStart := bar_index
            f_ev(bar_index, xC, "ST", "Secondary Test", d == 1, col, f_tip("Secondary Test", xC, volRel))
            f_phase(bar_index, "B", math.max(cLvl, bLvl), math.min(cLvl, bLvl))
            lastEv := "ST"
    else if st == 3
        bool sosNow = d * (close - bLvl) > 0.1 * atr and body >= sosBody * rng and d * (close - open) > 0 and (not hasVol or volZ >= sosZ)
        if pend > 0
            pend += 1
            if d * (pendExt - xC) > 0
                pendExt := xC
                pendBar := bar_index
                pendVol := math.max(pendVol, vol)
            if d * (cLvl - pendExt) > springPen * atr
                doBreak := true
            else if d * (close - cLvl) > 0
                spLvl    := pendExt
                spBar    := pendBar
                spVol    := pendVol
                doSpring := true
            else if pend > springBars
                doBreak := true
        else if d * (close - bLvl) > H
            endNow := true
            endTxt := "Left range"
        else if bar_index - bStart >= minB
            float pen = d * (cLvl - xC)
            if sosNow
                goSOS := true
            else if pen > 0.1 * atr
                if pen > springPen * atr
                    doBreak := true
                else if d * (close - cLvl) > 0
                    spLvl    := xC
                    spBar    := bar_index
                    spVol    := vol
                    doSpring := true
                else
                    pend    := 1
                    pendExt := xC
                    pendBar := bar_index
                    pendVol := vol
        else if d * (cLvl - close) > springPen * atr
            doBreak := true
    else if st == 4
        bool sosNow = d * (close - bLvl) > 0.1 * atr and body >= sosBody * rng and d * (close - open) > 0 and (not hasVol or volZ >= sosZ)
        if d * (spLvl - close) > 0
            endNow  := true
            endFail := true
            endTxt  := "Failed " + nmS
        else if d * (close - bLvl) > H
            endNow := true
            endTxt := "Left range"
        else if sosNow
            goSOS := true
        else if not hadTest and bar_index > spBar + 1 and d * (xC - cLvl) <= stZone * H and d * (xC - spLvl) >= 0 and (not hasVol or vol < spVol) and d * (close - open) >= 0
            doTest := true
    else if st == 5
        float mid   = (cLvl + bLvl) / 2
        bool  inLps = bar_index > sosBar and d * (xC - bLvl) <= lpsTol * atr and d * (xC - mid) > 0
        bool  conf  = d * (close - bLvl) > 0 and (d == 1 ? close - low >= 0.5 * rng : high - close >= 0.5 * rng)
        if inLps and conf
            doLPS := true
        else if d * (bLvl - close) > 0.5 * H
            st     := 3
            lastEv := "Back in range"
        else if d * (close - bLvl) >= H
            endNow := true
            endTxt := d == 1 ? "Markup (no LPS)" : "Markdown (no LPSY)"
        else if bar_index - sosBar > lpsMax
            endNow := true
            endTxt := d == 1 ? "No LPS" : "No LPSY"
    else if st == 6
        if not eShown and d * (close - bLvl) >= eMult * H
            eShown := true
            f_phase(bar_index, "E", math.max(cLvl, bLvl), math.min(cLvl, bLvl))
            lastEv := d == 1 ? "Markup" : "Markdown"
        if not (inTrade and tSch == schId) and not endNow
            if eShown or d * (bLvl - close) > 0.5 * H or bar_index - lpsBar > lpsMax * 2
                endNow := true
                endTxt := eShown ? (d == 1 ? "Markup" : "Markdown") : "Done"

    // ── break of the climax side: re-anchor (range extension) or fail ─
    if doBreak
        bool newClimax = d == 1 ? isSC : isBC
        if newClimax and reCnt < maxRe and math.abs(bLvl - xC) <= maxHAtr * atr
            reCnt     += 1
            nReAll    += 1
            cLvl      := xC
            cVol      := math.max(cVol, vol)
            cZv       := math.max(cZv, volZ)
            cRel      := volRel
            arBar     := bar_index
            st        := 2
            pend      := 0
            hadSpring := false
            hadTest   := false
            float top = math.max(cLvl, bLvl)
            float bot = math.min(cLvl, bLvl)
            if not na(rBox)
                box.set_top(rBox, top)
                box.set_bottom(rBox, bot)
                line.set_y1(midL, (top + bot) / 2)
                line.set_y2(midL, (top + bot) / 2)
            f_ev(bar_index, xC, d == 1 ? "SC" : "BC", d == 1 ? "Selling Climax (range extended)" : "Buying Climax (range extended)", d == 1, col, f_tip("Range extended to a new climax", xC, volRel))
            lastEv    := "Range extended"
        else
            endNow  := true
            endFail := true
            endTxt  := "Range broken"

    if doSpring
        float depth = d * (cLvl - spLvl) / atr
        int   spd   = bar_index - spBar
        float cl    = rng > 0 ? (d == 1 ? (close - low) / rng : (high - close) / rng) : 0.5
        int   q     = 0
        q += depth <= 0.5 ? 25 : depth <= 1.0 ? 15 : 5
        q += not hasVol ? 12 : spVol < cVol * 0.7 ? 25 : spVol < cVol ? 15 : 5
        q += spd == 0 ? 25 : spd == 1 ? 15 : 5
        q += cl >= 0.6 ? 25 : cl >= 0.4 ? 15 : 5
        spQ       := q
        hadSpring := true
        st        := 4
        pend      := 0
        nSp       += 1
        evSpring  := true
        f_ev(spBar, spLvl, nmS, d == 1 ? "Spring" : "Upthrust After Distribution", d == 1, col, f_tip(d == 1 ? "Spring" : "Upthrust After Distribution", spLvl, volRel[bar_index - spBar]) + "\nQuality: " + str.tostring(q) + "/100")
        f_phase(spBar, "C", math.max(cLvl, bLvl), math.min(cLvl, bLvl))
        lastEv    := nmS
        if aEvent

        evTest  := true
        lastEv  := "Test"

        int    score = f_score(cZv, hadSpring, spQ, true, 0.0, life, true, htf, d)
        string g     = f_grade(score)
        bool   sideOk = tradeSide == "Both" or (tradeSide == "Long only" and d == 1) or (tradeSide == "Short only" and d == -1)
        bool   ok    = entryMode != "LPS only" and not inTrade and sideOk and f_gIdx(g) <= f_gIdx(minGrade)
        f_ev(bar_index, xC, "Test" + (ok ? " · " + g : ""), "Test of " + nmS + (ok ? " · Grade " + g : ""), d == 1, col, f_tip("Test of " + nmS + " (lower effort)", xC, volRel) + "\nScore: " + str.tostring(score) + "/100")
        if aEvent
            msgs.push(f_msg("Test of " + nmS, d == 1 ? "long bias" : "short bias", close, ""))

            tBar    := bar_index
            tE      := close
            tSL     := spLvl - d * slAtr * atr

            tTP2    := bLvl + d * H
            tTP3    := bLvl + d * H * f_cause(life)
            tGrade  := g
            tScore  := score
            evEntry := true

            nSpSos += 1
        st     := 5
        sosBar := bar_index
        sosVol := vol
        sosZv  := volZ
        evSOS  := true
        f_ev(bar_index, xB, d == 1 ? "SOS" : "SOW", d == 1 ? "Sign of Strength" : "Sign of Weakness", d == -1, col, f_tip(d == 1 ? "Sign of Strength" : "Sign of Weakness", close, volRel))
        f_phase(bar_index, "D", math.max(cLvl, bLvl), math.min(cLvl, bLvl))
        lastEv := d == 1 ? "SOS" : "SOW"
        if aEvent
            msgs.push(f_msg(d == 1 ? "Sign of Strength" : "Sign of Weakness", d == 1 ? "long bias" : "short bias", close, ""))

    if doLPS
        int    life   = bar_index - cBar
        bool   lowEff = not hasVol or vol < sosVol

        string g      = f_grade(score)
        bool   sideOk = tradeSide == "Both" or (tradeSide == "Long only" and d == 1) or (tradeSide == "Short only" and d == -1)
        bool   ok     = entryMode != "Spring Test only" and not inTrade and sideOk and f_gIdx(g) <= f_gIdx(minGrade)
        string nmL    = d == 1 ? "LPS" : "LPSY"
        st        := 6
        lpsBar    := bar_index
        lastEv    := nmL
        lastScore := score
        f_ev(bar_index, xC, nmL + (ok ? " · " + g : ""), (d == 1 ? "Last Point of Support" : "Last Point of Supply") + (ok ? " · Grade " + g : ""), d == 1, col, f_tip(d == 1 ? "Last Point of Support" : "Last Point of Supply", xC, volRel) + "\nScore: " + str.tostring(score) + "/100")
        if ok
            float mid  = (cLvl + bLvl) / 2
            float slB  = slMode == "Range mid" ? mid : slMode == "LPS low / high" ? xC : (hadSpring ? spLvl : mid)
            inTrade := true
            tType   := nmL
            tDir    := d
            tBar    := bar_index
            tE      := close
            tSL     := slB - d * slAtr * atr
            tRisk   := math.abs(tE - tSL)
            tTP1    := bLvl + d * 0.5 * H
            tTP2    := bLvl + d * H
            tTP3    := bLvl + d * H * f_cause(life)
            tGrade  := g
            tScore  := score
            evEntry := true

    if evEntry

            tTP1 := tE + d * tRisk
        if d * (tTP2 - tTP1) < 0.5 * tRisk
            tTP2 := tTP1 + d * tRisk
        if d * (tTP3 - tTP2) < 0.5 * tRisk
            tTP3 := tTP2 + d * tRisk
        tCurSL  := tSL
        tSch    := schId
        tHits   := 0
        tReal   := 0.0
        tRemain := 1.0
        if showTrade
            tRiskB := box.new(bar_index, tE, bar_index + 1, tSL, border_color = color.new(distCol, 60), bgcolor = color.new(distCol, 85))
            qB.push(tRiskB)
            qBi.push(schId)
            tRewB := box.new(bar_index, tE, bar_index + 1, scaleOut ? tTP3 : tTP2, border_color = color.new(accCol, 60), bgcolor = color.new(accCol, 88))
            qB.push(tRewB)
            qBi.push(schId)
            tTp1L := line.new(bar_index, tTP1, bar_index + 1, tTP1, color = color.new(accCol, 0), style = line.style_dashed, width = 1)
            qN.push(tTp1L)
            qNi.push(schId)
            tTp2L := line.new(bar_index, tTP2, bar_index + 1, tTP2, color = color.new(accCol, 0), style = line.style_dashed, width = 1)
            qN.push(tTp2L)
            qNi.push(schId)
        if aEntry
            msgs.push(f_msg(tType + " entry · grade " + tGrade, d == 1 ? "long" : "short", tE, aJson ? ",\"sl\":" + str.tostring(tSL) + ",\"tp1\":" + str.tostring(tTP1) + ",\"tp2\":" + str.tostring(tTP2) + ",\"tp3\":" + str.tostring(tTP3) + ",\"score\":" + str.tostring(tScore) : " · SL " + str.tostring(tSL, format.mintick) + " · TP1 " + str.tostring(tTP1, format.mintick) + " · TP2 " + str.tostring(tTP2, format.mintick) + " · TP3 " + str.tostring(tTP3, format.mintick)))

    // ── global expiry ──────────────────────────────────────────────
    if not endNow and st >= 1 and st <= 5 and bar_index - cBar > maxLife
        endNow := true
        endTxt := "Range expired"

    // ── extend / close the active range ────────────────────────────
    if st >= 2 and not na(rBox)
        box.set_right(rBox, bar_index)
        line.set_x2(midL, bar_index)
        if not endNow
            box.set_text(rBox, (d == 1 ? "Accumulation" : "Distribution") + " · Phase " + f_phaseName(st))
    if endNow
        if st >= 2
            if not na(rBox)
                box.set_text(rBox, (d == 1 ? "Accumulation" : "Distribution") + " · " + endTxt)
            if endFail
                nFail  += 1
                evFail := true
                if aFail
                    msgs.push(f_msg(endTxt, "", close, ""))
                if not na(rBox)
                    box.set_bgcolor(rBox, color.new(color.gray, 90))
                    box.set_border_color(rBox, color.new(color.gray, 40))
                    box.set_text_color(rBox, color.new(color.gray, 10))
                if showFail
                    f_ev(bar_index, close, "✕", "✕ " + endTxt, d == 1, color.new(#F23645, 0), endTxt)
        lastEv := endTxt
        st     := 0
        pend   := 0

"Retest of climax zone on lower volume (ST)" : st == 3 ? (bar_index - bStart < minB ? "Test on lower volume or " + nO : st == 5 ? nL + " pullback to " + str.tostring(bLvl, format.mintick) + " (" + str.tostring(math.max(lpsMax - (bar_index - sosBar), 0)) + " bars left)" : (inTrade ? "Targets" : (d == 1 ? "Markup" : "Markdown"))

"Demand drying up" : "Demand still active") : st == 4 ? "Late buyers trapped" : st == 5 ? "Supply stepping in" : "Supply in control")

if barstate.islast
